//AOE//
static float aoeval = 0.0f;
DWORD jmpbackaoe;
void __declspec(naked) AOECave()
{
	__asm
	{
		fadd dword ptr[aoeval]
		fcompp
		fnstsw ax
		test ah, 0x41
		jmp[jmpbackaoe]
	}
}
//LR//
static float lrval = 0.0f;
DWORD jmpbacklr;
void __declspec(naked) LRCave()
{
	__asm
	{
		mov eax, [edx + 0x10]
		fsub dword ptr[lrval]
		fstp dword ptr[esp + 0x30]
		jmp[jmpbacklr]
	}
}
//AS//
static float asval = 0.0f;
DWORD jmpbackas;
void __declspec(naked) ASCave()
{
	__asm
	{
		fld dword ptr[ecx + 0x00004730]
		fadd dword ptr[asval]
		jmp[jmpbackas]
	}
}
//CS//
